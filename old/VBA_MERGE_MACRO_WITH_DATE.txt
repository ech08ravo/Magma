========================================
VBA MACRO: Curriculum Data Merge Tool
(Output to Magma Folder with Date)
========================================

INSTRUCTIONS:
1. Open your Excel file with both sheets: "Courses by Units" and "Units by Assessment Tasks"
2. Press Cmd+F11 (Mac) or Alt+F11 (Windows) to open Visual Basic Editor
3. Insert a new Module (Insert > Module)
4. Copy the ENTIRE code below into the module
5. Close the editor and save as .xlsm (Excel Macro-Enabled Workbook)
6. Run: Tools > Macros > Run Macro > MergeCurriculumWithDate
7. CSV will be created in Magma folder with today's date

OUTPUT FILENAME: curriculum-data-YYYYMMDD.csv
Example: curriculum-data-20260501.csv

========================================
COPY THIS CODE INTO VBA MODULE:
========================================

Sub MergeCurriculumWithDate()
    Dim wsUnits As Worksheet, wsAssessment As Worksheet
    Dim i As Long, j As Long, k As Long
    Dim lastRowUnits As Long, lastRowAssessment As Long
    Dim unitCode As String, pathwayId As String
    Dim outputRow As String
    Dim unitLevel As String, teachingPeriods As String
    Dim atNumber As String, atTitle As String, studentOutput As String
    Dim aoLType As String, mode As String, participation As String
    Dim hurdle As String, weighting As String
    Dim unitGLOString As String, taskGLOString As String
    Dim assessmentUnitCode As String
    Dim csvFile As String
    Dim rowCount As Long
    Dim fileNumber As Integer
    Dim magmaFolder As String
    Dim dateStr As String

    On Error GoTo ErrorHandler

    ' Get worksheets - MUST match sheet names exactly
    Set wsUnits = ThisWorkbook.Sheets("Courses by Units")
    Set wsAssessment = ThisWorkbook.Sheets("Units by Assessment Tasks")

    ' Get last row of each sheet (using -4162 for xlUp - Mac compatible)
    lastRowUnits = wsUnits.Cells(wsUnits.Rows.Count, 1).End(-4162).Row
    lastRowAssessment = wsAssessment.Cells(wsAssessment.Rows.Count, 1).End(-4162).Row

    ' Create file path in Magma folder with date
    ' Magma folder: ~/Library/Mobile Documents/com~apple~CloudDocs/Debusseycat/Magma/
    magmaFolder = Environ("HOME") & "/Library/Mobile Documents/com~apple~CloudDocs/Debusseycat/Magma"

    ' Generate date string: YYYYMMDD format
    dateStr = Format(Today, "yyyymmdd")

    ' Build full file path
    csvFile = magmaFolder & "/curriculum-data-" & dateStr & ".csv"

    ' Get free file number for writing
    fileNumber = FreeFile

    ' Open CSV file for writing
    Open csvFile For Output As fileNumber

    ' Write header row
    Print #fileNumber, "pathway_id,unit_level,unit_code,teaching_periods,at_number,at_title,student_output,aol_task_type,mode,participation,hurdle,weighting,unit_glo1,unit_glo2,unit_glo3,unit_glo4,unit_glo5,unit_glo6,unit_glo7,unit_glo8,task_glo1,task_glo2,task_glo3,task_glo4,task_glo5,task_glo6,task_glo7,task_glo8"

    rowCount = 0

    ' ===== MAIN MERGE LOOP =====
    ' For each unit in Courses by Units
    For i = 2 To lastRowUnits
        pathwayId = CStr(wsUnits.Cells(i, 1).Value)

        ' Skip empty rows
        If pathwayId <> "" Then
            unitCode = CStr(wsUnits.Cells(i, 3).Value)

            ' Skip rows without unit code
            If unitCode <> "" Then

                ' Get unit-level data
                unitLevel = CStr(wsUnits.Cells(i, 2).Value)
                teachingPeriods = CStr(wsUnits.Cells(i, 4).Value)

                ' Get unit GLO values (columns 5-12 = E-L)
                unitGLOString = ""
                For j = 5 To 12
                    unitGLOString = unitGLOString & CStr(wsUnits.Cells(i, j).Value)
                    If j < 12 Then unitGLOString = unitGLOString & ","
                Next j

                ' ===== INNER LOOP: Find all assessment tasks for this unit =====
                For k = 2 To lastRowAssessment
                    assessmentUnitCode = CStr(wsAssessment.Cells(k, 1).Value)

                    ' Only process if unit code matches
                    If assessmentUnitCode = unitCode Then

                        ' Get assessment task data (columns A-I, then J-Q)
                        atNumber = CStr(wsAssessment.Cells(k, 2).Value)
                        atTitle = QuoteCSV(CStr(wsAssessment.Cells(k, 3).Value))
                        studentOutput = QuoteCSV(CStr(wsAssessment.Cells(k, 4).Value))
                        aoLType = QuoteCSV(CStr(wsAssessment.Cells(k, 5).Value))
                        mode = QuoteCSV(CStr(wsAssessment.Cells(k, 6).Value))
                        participation = QuoteCSV(CStr(wsAssessment.Cells(k, 7).Value))
                        hurdle = CStr(wsAssessment.Cells(k, 8).Value)
                        weighting = QuoteCSV(CStr(wsAssessment.Cells(k, 9).Value))

                        ' Get task GLO values (columns 10-17 = J-Q)
                        taskGLOString = ""
                        For j = 10 To 17
                            taskGLOString = taskGLOString & CStr(wsAssessment.Cells(k, j).Value)
                            If j < 17 Then taskGLOString = taskGLOString & ","
                        Next j

                        ' Build combined output row
                        outputRow = pathwayId & "," & _
                                   unitLevel & "," & _
                                   unitCode & "," & _
                                   QuoteCSV(teachingPeriods) & "," & _
                                   atNumber & "," & _
                                   atTitle & "," & _
                                   studentOutput & "," & _
                                   aoLType & "," & _
                                   mode & "," & _
                                   participation & "," & _
                                   hurdle & "," & _
                                   weighting & "," & _
                                   unitGLOString & "," & _
                                   taskGLOString

                        ' Write row to CSV
                        Print #fileNumber, outputRow
                        rowCount = rowCount + 1
                    End If
                Next k
            End If
        End If
    Next i

    ' Close file
    Close fileNumber

    MsgBox "✓ Success!" & vbCrLf & vbCrLf & _
           "CSV file created in Magma folder:" & vbCrLf & _
           "curriculum-data-" & dateStr & ".csv" & vbCrLf & vbCrLf & _
           "Rows written: " & rowCount & vbCrLf & vbCrLf & _
           "Ready for dashboard.", _
           vbInformation, "Done"

    Exit Sub

ErrorHandler:
    ' Clean up file if error occurs
    On Error Resume Next
    Close fileNumber
    On Error GoTo 0

    MsgBox "Error: " & Err.Description & vbCrLf & vbCrLf & _
           "CHECKLIST:" & vbCrLf & _
           "☐ Sheet 1 named 'Courses by Units'?" & vbCrLf & _
           "☐ Sheet 2 named 'Units by Assessment Tasks'?" & vbCrLf & _
           "☐ Data starts in row 2?" & vbCrLf & _
           "☐ Magma folder exists at:" & vbCrLf & _
           "  ~/Library/Mobile Documents/" & vbCrLf & _
           "  com~apple~CloudDocs/Debusseycat/Magma/", vbCritical, "Error"
End Sub

' Helper function to properly quote CSV fields
Function QuoteCSV(value As String) As String
    Dim needsQuote As Boolean

    needsQuote = False

    ' Check if value contains comma, quotes, or newlines
    If InStr(value, ",") > 0 Then needsQuote = True
    If InStr(value, """") > 0 Then needsQuote = True
    If InStr(value, vbCrLf) > 0 Then needsQuote = True
    If InStr(value, vbCr) > 0 Then needsQuote = True
    If InStr(value, vbLf) > 0 Then needsQuote = True

    If needsQuote Then
        ' Escape quotes by doubling them
        QuoteCSV = """" & Replace(value, """", """""") & """"
    Else
        QuoteCSV = value
    End If
End Function
